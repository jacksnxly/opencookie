import {
  freezeConsentRecord,
  snapshotCurrentConsentRecord,
  type ConsentRecord,
} from "./consent-record.js";
import { freezeIssues } from "./record-decision.js";

/** @public */
export interface InvalidWithdrawalTimeIssue {
  readonly code: "invalid-withdrawal-time";
  readonly path: "withdrawnAt";
}

/** @public */
export interface InvalidWithdrawalRecordIssue {
  readonly code: "invalid-current-record";
  readonly path:
    | "current"
    | "current.consentId"
    | "current.decidedAt"
    | "current.decisions"
    | "current.expiresAt"
    | "current.locale"
    | "current.method"
    | "current.policy"
    | "current.recordId"
    | "current.revision"
    | "current.schemaVersion"
    | "current.withdrawnAt";
}

/** @public */
export type WithdrawConsentResult =
  | {
      readonly issues: readonly (
        InvalidWithdrawalRecordIssue | InvalidWithdrawalTimeIssue
      )[];
      readonly ok: false;
    }
  | {
      readonly ok: true;
      readonly outcome: "changed";
      readonly previousRecord: ConsentRecord;
      readonly record: ConsentRecord;
    }
  | {
      readonly ok: true;
      readonly outcome: "unchanged";
      readonly record: ConsentRecord;
    };

function rejected(
  issue: InvalidWithdrawalRecordIssue | InvalidWithdrawalTimeIssue,
): WithdrawConsentResult {
  return Object.freeze({
    issues: freezeIssues([issue]),
    ok: false,
  });
}

/**
 * Resolves every existing grant to a denial. This transition governs future
 * access and deliberately makes no assertion about past transmissions.
 *
 * @public
 */
export function withdrawConsent(
  current: ConsentRecord,
  withdrawnAt: number,
): WithdrawConsentResult {
  const snapshot = snapshotCurrentConsentRecord(current);
  if (!snapshot.ok) {
    return rejected({
      code: "invalid-current-record",
      path: snapshot.path,
    });
  }
  const currentRecord = snapshot.record;

  const currentTransitionAt =
    currentRecord.method === "withdrawal"
      ? currentRecord.withdrawnAt
      : currentRecord.decidedAt;
  if (!Number.isSafeInteger(withdrawnAt) || withdrawnAt < currentTransitionAt) {
    return rejected({
      code: "invalid-withdrawal-time",
      path: "withdrawnAt",
    });
  }

  const hasGrant = currentRecord.decisions.some(
    ({ value }) => value === "granted",
  );
  if (!hasGrant) {
    return Object.freeze({
      ok: true,
      outcome: "unchanged",
      record: currentRecord,
    });
  }

  const record = freezeConsentRecord({
    ...currentRecord,
    decisions: currentRecord.decisions.map((decision) => ({
      purposeId: decision.purposeId,
      value: "denied",
    })),
    method: "withdrawal",
    revision: currentRecord.revision + 1,
    withdrawnAt,
  });

  return Object.freeze({
    ok: true,
    outcome: "changed",
    previousRecord: currentRecord,
    record,
  });
}
