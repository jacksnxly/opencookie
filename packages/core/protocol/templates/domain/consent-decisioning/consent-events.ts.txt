import { diffDecisions, type DecisionDiff } from "./diff-decisions.js";
import {
  freezeIdentity,
  snapshotConsentRecord,
  type ConsentPolicyIdentityInput,
  type ConsentRecord,
} from "./consent-record.js";

/** @public */
export interface ConsentEventBatchInput {
  readonly committed: boolean;
  readonly previousRecord: ConsentRecord | null;
  readonly record: ConsentRecord;
}

/** @public */
export interface ConsentRecordCreatedEvent {
  readonly occurredAt: number;
  readonly policy: ConsentPolicyIdentityInput;
  readonly recordId: string;
  readonly revision: number;
  readonly type: "ConsentRecordCreated";
}

/** @public */
export interface ConsentDecisionEvent {
  readonly changes: readonly DecisionDiff[];
  readonly occurredAt: number;
  readonly policy: ConsentPolicyIdentityInput;
  readonly recordId: string;
  readonly revision: number;
  readonly type:
    "ConsentChanged" | "ConsentDenied" | "ConsentGranted" | "ConsentWithdrawn";
}

/** @public */
export type ConsentDomainEvent =
  ConsentDecisionEvent | ConsentRecordCreatedEvent;

function transitionOccurredAt(record: ConsentRecord): number {
  return record.method === "withdrawal" ? record.withdrawnAt : record.decidedAt;
}

function recordCreated(record: ConsentRecord): ConsentRecordCreatedEvent {
  return Object.freeze({
    occurredAt: record.decidedAt,
    policy: freezeIdentity(record.policy),
    recordId: record.recordId,
    revision: record.revision,
    type: "ConsentRecordCreated",
  });
}

function decisionEvent(
  record: ConsentRecord,
  changes: readonly DecisionDiff[],
  type: ConsentDecisionEvent["type"],
): ConsentDecisionEvent {
  return Object.freeze({
    changes,
    occurredAt:
      record.method === "withdrawal" ? record.withdrawnAt : record.decidedAt,
    policy: freezeIdentity(record.policy),
    recordId: record.recordId,
    revision: record.revision,
    type,
  });
}

/**
 * Creates the domain-event batch only after application orchestration marks a
 * transition committed. A failed grant candidate therefore emits nothing.
 *
 * @public
 */
export function createConsentEventBatch(
  input: ConsentEventBatchInput,
): readonly ConsentDomainEvent[] {
  if (!input.committed) return Object.freeze([]);
  const recordSnapshot = snapshotConsentRecord(input.record);
  if (!recordSnapshot.ok) return Object.freeze([]);
  const record = recordSnapshot.record;
  let previousRecord: ConsentRecord | null = null;
  if (input.previousRecord !== null) {
    const previousSnapshot = snapshotConsentRecord(input.previousRecord);
    if (!previousSnapshot.ok) return Object.freeze([]);
    previousRecord = previousSnapshot.record;
  }
  if (!previousRecord) {
    if (record.revision !== 1) return Object.freeze([]);
  } else if (
    record.recordId !== previousRecord.recordId ||
    record.revision !== previousRecord.revision + 1
  ) {
    return Object.freeze([]);
  }
  if (
    previousRecord &&
    transitionOccurredAt(record) < transitionOccurredAt(previousRecord)
  ) {
    return Object.freeze([]);
  }

  const changes = diffDecisions(
    previousRecord?.decisions ?? null,
    record.decisions,
  );
  if (!previousRecord) {
    const eventType = record.decisions.some(({ value }) => value === "granted")
      ? "ConsentGranted"
      : "ConsentDenied";

    return Object.freeze([
      recordCreated(record),
      decisionEvent(record, changes, eventType),
    ]);
  }

  return Object.freeze([
    decisionEvent(
      record,
      changes,
      record.method === "withdrawal" ? "ConsentWithdrawn" : "ConsentChanged",
    ),
  ]);
}
