import {
  comparePurposeId,
  stableUnique,
  type ConsentDecisionState,
  type PurposeDecisionState,
} from "./consent-record.js";

/** @public */
export interface DecisionDiff {
  readonly from: ConsentDecisionState;
  readonly purposeId: string;
  readonly to: ConsentDecisionState;
}

function assertValidDecisionSet(
  label: "current" | "previous",
  decisions: readonly PurposeDecisionState[],
): void {
  const seen = new Set<string>();

  for (const [index, decision] of decisions.entries()) {
    const purposeId: unknown = decision.purposeId;
    if (typeof purposeId !== "string" || purposeId.trim().length === 0) {
      throw new RangeError(
        `${label} contains invalid purpose identifier at index ${String(index)}`,
      );
    }
    if (seen.has(purposeId)) {
      throw new RangeError(
        `${label} contains duplicate purpose decision: ${purposeId}`,
      );
    }
    seen.add(purposeId);

    const value: unknown = decision.value;
    if (value !== "denied" && value !== "granted" && value !== "unresolved") {
      throw new RangeError(
        `${label} contains invalid purpose decision: ${decision.purposeId}`,
      );
    }
  }
}

/**
 * Computes purpose transitions without depending on object or input order.
 * A missing prior record represents the unresolved first-visit state.
 *
 * @public
 */
export function diffDecisions(
  previous: readonly PurposeDecisionState[] | null,
  current: readonly PurposeDecisionState[],
): readonly DecisionDiff[] {
  if (previous) assertValidDecisionSet("previous", previous);
  assertValidDecisionSet("current", current);

  const previousByPurpose = new Map(
    (previous ?? []).map((decision) => [decision.purposeId, decision.value]),
  );
  const currentByPurpose = new Map(
    current.map((decision) => [decision.purposeId, decision.value]),
  );
  const purposeIds = stableUnique([
    ...previousByPurpose.keys(),
    ...currentByPurpose.keys(),
  ]);
  const changes = purposeIds.flatMap((purposeId) => {
    const from = previousByPurpose.get(purposeId) ?? "unresolved";
    const to = currentByPurpose.get(purposeId) ?? "unresolved";
    return from === to
      ? []
      : [
          Object.freeze({
            from,
            purposeId,
            to,
          }),
        ];
  });

  return Object.freeze(changes.sort(comparePurposeId));
}
