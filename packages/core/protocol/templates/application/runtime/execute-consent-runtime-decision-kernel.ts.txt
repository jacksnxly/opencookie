/** @public */
export interface PendingGrantRuntimeTransition {
  readonly persistenceStatus: "pending";
  readonly record: ConsentRecord;
  readonly transitionId: string;
}

/** @public */
export type ConsentRuntimeTransition =
  | DenialTransition
  | GrantTransition
  | PendingDenialTransition
  | PendingGrantRuntimeTransition;

/** @public */
export interface ConsentRuntimeTransitionEffect {
  readonly events: readonly ConsentDomainEvent[];
  readonly persistenceStatus:
    "confirmed" | "failed" | "failed-conservative" | "pending";
  readonly record: ConsentRecord;
  readonly runtime: ConsentRuntimeProjection;
  readonly transitionId: string;
}

/** @public */
export interface ConsentRuntimeDecisionCommand {
  readonly currentRecord?: ConsentRecord;
  readonly decision: RecordDecisionCommand;
  readonly now: number;
  /** Durable base used only while retrying an unconfirmed transition. */
  readonly persistedRecord?: ConsentRecord | null;
  readonly policyHistory?: readonly PolicySnapshot[];
  readonly policySnapshot: PolicySnapshot;
  readonly privacyRestrictions?: ProcessingRestrictionSet;
  readonly resourcePolicy: DefinedResourcePolicy;
  readonly servicePreferences?: ServicePreferenceRecord;
  readonly transitionId: string;
}

/** @public */
export interface DenialReceiptDiagnostic {
  readonly code: "denial-receipt-failed";
}

/** @public */
export type ConsentRuntimeDecisionDiagnostic =
  | DenialTransitionDiagnostic
  | DenialReceiptDiagnostic
  | GrantPersistenceDiagnostic
  | GrantActivationDiagnostic;

/** @public */
export type ConsentRuntimeDecisionResult =
  | {
      readonly issues: readonly RecordDecisionIssue[];
      readonly outcome: "rejected";
    }
  | {
      readonly outcome: "unchanged";
      readonly record: ConsentRecord;
      readonly runtime: ConsentRuntimeProjection;
    }
  | {
      readonly effect: ConsentRuntimeTransitionEffect;
      readonly outcome: "committed";
      readonly transition: DenialTransition | GrantTransition;
    }
  | {
      readonly diagnostics: readonly ConsentRuntimeDecisionDiagnostic[];
      readonly effect: ConsentRuntimeTransitionEffect;
      readonly outcome: "degraded";
      readonly transition: DenialTransition | GrantTransition;
    }
  | {
      readonly diagnostics?: readonly ConsentRuntimeDecisionDiagnostic[];
      readonly effect: ConsentRuntimeTransitionEffect;
      readonly outcome: "pending";
      readonly transition: PendingGrantRuntimeTransition;
    }
  | {
      readonly diagnostics: readonly ConsentRuntimeDecisionDiagnostic[];
      readonly effect: ConsentRuntimeTransitionEffect;
      readonly outcome: "invalidated";
      readonly transition: PendingGrantRuntimeTransition;
    }
  | {
      readonly diagnostics: readonly [
        { readonly code: "grant-superseded" },
      ];
      readonly outcome: "invalidated";
      readonly stage: "intent";
    };

function assertRuntimeTime(decidedAt: number, now: number): void {
  if (!Number.isSafeInteger(now)) {
    throw new TypeError("now must be a finite safe integer.");
  }
  if (now < decidedAt) {
    throw new RangeError("now must be greater than or equal to decidedAt.");
  }
}

/**
 * Receipt-neutral decision kernel shared by the public executor facade and
 * the package-private client kernels. Ports are supplied through a capture
 * closure invoked at the exact position the public executor validated its
 * hostile inputs, so rejected and unchanged commands never observe ports.
 */
export async function executeConsentRuntimeDecisionKernel(
  snapshotPorts: () => CapturedConsentRuntimeKernelPorts,
  command: ConsentRuntimeDecisionCommand,
  context: ConsentRuntimeProjectionContext,
): Promise<ConsentRuntimeDecisionResult> {
  const now = command.now;
  const consentPolicy = projectConsentPolicy(context.policySnapshot);
  const transitionId = command.transitionId;
  const persistedRecord = command.persistedRecord;
  const decision = recordConsentDecision(
    command.currentRecord,
    consentPolicy,
    command.decision,
  );
  if (!decision.ok) {
    return Object.freeze({ issues: decision.issues, outcome: "rejected" });
  }
  const currentRecord =
    decision.outcome === "changed" ? decision.previousRecord : decision.record;

  const persistenceRetry =
    decision.outcome === "unchanged" && persistedRecord !== undefined;
  assertRuntimeTime(decision.record.decidedAt, now);
  const receiptOccurredAt = persistenceRetry ? now : decision.record.decidedAt;
  if (decision.outcome === "unchanged" && !persistenceRetry) {
    return Object.freeze({
      outcome: "unchanged",
      record: decision.record,
      runtime: projectRuntimeFor(context, decision.record, now),
    });
  }

  const persistedBase =
    persistedRecord === undefined ? currentRecord : persistedRecord;

  const prepareEffect = (record: ConsentRecord): PreparedRuntimeEffect =>
    Object.freeze({
      events: createConsentEventBatch({
        committed: true,
        previousRecord: currentRecord,
        record,
      }),
      record,
      runtime: projectRuntimeFor(context, record, now),
    });

  const persisted = toPersistedDecisionRecord(decision.record);
  const confirmedEffect = prepareEffect(toConsentRecord(persisted));
  const isCompleteDenial = Object.values(persisted.decisions).every(
    (value) => value === "denied",
  );
  const conservativeEffect = isCompleteDenial
    ? undefined
    : prepareEffect(
        toConsentRecord(
          projectConservativePersistenceFailure(
            currentRecord === null
              ? null
              : toPersistedDecisionRecord(currentRecord),
            persisted,
          ).effectiveRecord,
        ),
      );
  const capturedPorts = snapshotPorts();
  const [transitionPorts, effectFor] = createRuntimeTransitionProjection(
    capturedPorts,
    confirmedEffect,
    conservativeEffect,
    receiptOccurredAt,
  );

  if (isCompleteDenial) {
    const result = await (
      await executeDenialTransitionProgram({
        command: {
          expectedRecord:
            persistedBase === null
              ? null
              : toPersistedDecisionRecord(persistedBase),
          rejection: persisted,
          transitionId,
        },
        degradeReceiptFailure: true,
        kind: "ordinary",
        ports: transitionPorts,
      })
    ).completion;
    return projectRuntimeTransitionResult(result, effectFor);
  }

  const result = await executeGrantTransition(transitionPorts, {
    candidate: persisted,
    expectedRecord:
      persistedBase === null ? null : toPersistedDecisionRecord(persistedBase),
    previous:
      currentRecord === null ? null : toPersistedDecisionRecord(currentRecord),
    transitionId,
  });
  return projectRuntimeTransitionResult(result, effectFor);
}
